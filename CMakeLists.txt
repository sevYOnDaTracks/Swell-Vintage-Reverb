cmake_minimum_required(VERSION 3.22)
project(SwellReverb VERSION 0.1.0 LANGUAGES C CXX)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
set(REVERB_FORMATS VST3 Standalone)
if(APPLE)
  list(APPEND REVERB_FORMATS AU)
endif()
# Never change PLUGIN_CODE / BUNDLE_ID once released: saved DAW sessions would lose the plugin.
juce_add_plugin(SwellReverb COMPANY_NAME "Swell-audio" BUNDLE_ID "fr.swell-audio.reverb" PLUGIN_MANUFACTURER_CODE Swel PLUGIN_CODE Srvb FORMATS ${REVERB_FORMATS} PRODUCT_NAME "Swell Reverb" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
juce_add_binary_data(ReverbAssets SOURCES swell-mark.png)
target_sources(SwellReverb PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(SwellReverb PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(SwellReverb PRIVATE ReverbAssets juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
enable_testing()
juce_add_console_app(ReverbTest PRODUCT_NAME "Reverb Audio Tests")
target_sources(ReverbTest PRIVATE Tests/AudioTest.cpp)
target_include_directories(ReverbTest PRIVATE "${juce_SOURCE_DIR}/modules")
target_compile_definitions(ReverbTest PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
target_link_libraries(ReverbTest PRIVATE SwellReverb)
add_test(NAME ReverbAudio COMMAND ReverbTest)
juce_add_console_app(ReverbSnapshot PRODUCT_NAME "Reverb Snapshot")
target_sources(ReverbSnapshot PRIVATE Tools/Snapshot.cpp)
target_include_directories(ReverbSnapshot PRIVATE "${juce_SOURCE_DIR}/modules")
target_compile_definitions(ReverbSnapshot PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
target_link_libraries(ReverbSnapshot PRIVATE SwellReverb)
juce_add_console_app(ReverbDemo PRODUCT_NAME "Reverb Demo Renderer")
target_sources(ReverbDemo PRIVATE Tools/RenderDemo.cpp)
target_include_directories(ReverbDemo PRIVATE "${juce_SOURCE_DIR}/modules")
target_compile_definitions(ReverbDemo PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
target_link_libraries(ReverbDemo PRIVATE SwellReverb)
